cmake_minimum_required(VERSION 3.25)
project(SonicLinuxHostContracts LANGUAGES CXX)
get_filename_component(SONIC_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(SONIC_LINUX_SDK "${SONIC_ROOT}/.local/linux-sdk")
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
find_package(Threads REQUIRED)
add_library(SDL3::SDL3 SHARED IMPORTED)
set_target_properties(SDL3::SDL3 PROPERTIES IMPORTED_LOCATION "${SONIC_ROOT}/build-linux/sdl/libSDL3.so.0.4.16"
    INTERFACE_INCLUDE_DIRECTORIES "${SONIC_ROOT}/.local/linux-deps/sdl/SDL3-3.4.16/include")
add_library(sonic_linux_sdk_headers INTERFACE)
target_include_directories(sonic_linux_sdk_headers INTERFACE "${SONIC_LINUX_SDK}/include"
    "${SONIC_ROOT}/.local/baseline/r354/sdk/generated/include")
include("${SONIC_ROOT}/cmake/LinuxPlatform.cmake")
include("${SONIC_ROOT}/cmake/LinuxAudio.cmake")
add_executable(sonic-linux-platform-tests "${SONIC_ROOT}/tools/test_linux_platform.cpp"
    "${SONIC_LINUX_SDK}/src/runtime/native_port.cpp")
target_compile_options(sonic-linux-platform-tests PRIVATE -O2 -g0 -ffunction-sections -fdata-sections)
target_link_options(sonic-linux-platform-tests PRIVATE -Wl,--gc-sections)
target_link_libraries(sonic-linux-platform-tests PRIVATE sonic_linux_platform)
set_target_properties(sonic-linux-platform-tests PROPERTIES BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "$ORIGIN/lib")
add_executable(sonic-linux-input-tests "${SONIC_ROOT}/tools/test_input_mapping.cpp"
    "${SONIC_ROOT}/src/sonic_input.cpp" "${SONIC_ROOT}/src/sonic_presentation.cpp")
target_include_directories(sonic-linux-input-tests PRIVATE "${SONIC_ROOT}/src")
target_compile_options(sonic-linux-input-tests PRIVATE -O2 -g0 -ffunction-sections -fdata-sections)
target_link_options(sonic-linux-input-tests PRIVATE -Wl,--gc-sections)
target_link_libraries(sonic-linux-input-tests PRIVATE sonic_linux_platform)
set_target_properties(sonic-linux-input-tests PROPERTIES BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "$ORIGIN/lib")
add_executable(sonic-linux-audio-tests "${SONIC_ROOT}/tools/test_linux_audio.cpp")
target_include_directories(sonic-linux-audio-tests PRIVATE "${SONIC_ROOT}/src")
target_compile_options(sonic-linux-audio-tests PRIVATE -O2 -g0 -ffunction-sections -fdata-sections)
target_link_options(sonic-linux-audio-tests PRIVATE -Wl,--gc-sections)
target_link_libraries(sonic-linux-audio-tests PRIVATE sonic_linux_audio)
set_target_properties(sonic-linux-audio-tests PROPERTIES BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "$ORIGIN/lib")
